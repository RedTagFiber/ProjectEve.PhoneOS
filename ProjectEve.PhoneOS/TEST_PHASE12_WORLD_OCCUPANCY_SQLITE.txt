SELECT NpcId,HomeLocationId,WorkLocationId,ScheduleMode

SELECT NpcId,NpcName,Status,CurrentLocationId,
       OriginLocationId,DestinationLocationId,
       DepartGameTime,ExpectedArrivalGameTime,Activity,Source,UpdatedGameTime
FROM NpcWorldLocationState

SELECT Id,NpcId,Kind,StartGameTime,EndGameTime,LocationId,Status,Note
FROM NpcScheduleOverride

FROM NpcShiftAssignment

SELECT Id,NpcId,NpcName,FromStatus,ToStatus,
       FromLocationId,ToLocationId,GameTime,Source
FROM NpcWorldMovementEvent
