OPTIONAL SQLITE INSPECTION

SELECT PlayerId,PlayerName,LocationId,LocationDisplayName,SceneId,
       Status,Activity,Attention,XFeet,YFeet,FacingDegrees,
       HasSpatialSnapshot,ClientAttached,
       LastClientHeartbeatRealUtc,UpdatedGameTime
FROM PlayerWorldPresenceState;

SELECT Id,PlayerId,PlayerName,FromLocationId,ToLocationId,
       FromSceneId,ToSceneId,GameTime,Reason
FROM PlayerWorldMovementEvent
ORDER BY Id DESC;

SELECT SceneId,PlayerId,PlayerName,SlotIndex,LastHeartbeatRealUtc
FROM SharedScenePlayerMembership
ORDER BY SceneId,SlotIndex;

SELECT SceneId,CharacterKey,PlayerId,DisplayName,
       XFeet,YFeet,FacingDegrees,Attention,Activity,IsActive
FROM ScenePresence
WHERE IsPlayer=1
ORDER BY SceneId,CharacterKey;
