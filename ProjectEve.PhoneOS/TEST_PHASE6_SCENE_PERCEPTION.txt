-- PROJECT EVE PHASE 6 — SCENE / PERCEPTION DIAGNOSTICS
-- Run against the same project_eve.db used by the running server.

-- 1) Current active scenes
SELECT SceneId, LocationId, DisplayName, AmbientNoise, VisualClutter,
       UpdatedGameTime, UpdatedRealUtc
FROM ActiveScene
ORDER BY UpdatedRealUtc DESC;

-- 2) Physical world truth: who is in the newest scene and where.
SELECT p.SceneId, p.CharacterKey, p.NpcId, p.PlayerId, p.DisplayName,
       p.XFeet, p.YFeet, p.RoomZone, p.AcousticZone,
       p.Attention, p.Activity, p.Concealment, p.IsActive
FROM ScenePresence p
WHERE p.SceneId=(SELECT SceneId FROM ActiveScene ORDER BY UpdatedRealUtc DESC LIMIT 1)
ORDER BY p.IsPlayer DESC, p.DisplayName;

-- 3) What observers ACTUALLY perceived.
SELECT Id, GameTime, EventKind, SourceCharacterKey, ObserverCharacterKey,
       Quality, printf('%.2f', Confidence) AS Confidence,
       printf('%.1f', DistanceFeet) AS Feet,
       PerceivedText
FROM ScenePerceptionEvidence
ORDER BY Id DESC
LIMIT 60;

     XFeet,YFeet,FacingDegrees,RoomZone,AcousticZone,
     Attention,Activity,Concealment,IsActive,UpdatedGameTime,UpdatedRealUtc)

    XFeet=8,
    YFeet=0,
    Attention=0.90,
    Activity='idle',
    Concealment=0.95,
    IsActive=1,
    UpdatedRealUtc=datetime('now');

-- After reloading the page and saying several normal sentences to Eve:
SELECT Id, GameTime, EventKind, SourceCharacterKey, ObserverCharacterKey,
       Quality, printf('%.2f', Confidence) AS Confidence,
       printf('%.1f', DistanceFeet) AS Feet,

-- 4) Exact conversation evidence vs NPC perception overlay.
-- Exact text remains in ConversationMessage. The overlay is what the NPC participant
-- is allowed to use when it did not fully hear/see that player turn.
SELECT m.SessionId,m.Sequence,m.Role,m.Speaker,m.MessageText AS ExactMessage,
       p.ObserverNpcId,p.PerceivedText,p.SourceEventKey
FROM ConversationMessage m
LEFT JOIN ConversationMessagePerception p ON p.MessageId=m.Id
ORDER BY m.Id DESC
LIMIT 40;
